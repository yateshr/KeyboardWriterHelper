# Typing tests, run by test.cmd:  KeyboardWriterHelper.exe --cases tests\typing-cases.txt
#
#     "what is typed" => "what must be on screen"
#
# {BS} is Backspace, {NEXT} the next-suggestion shortcut, {CLICK} a mouse click
# (the caret may have moved), {ENTER} Enter. A line starting with @
# sets up the cases below it:
#     @ word tarun               add to My words
#     @ fix brb=be right back    add to My fixes
#     @ punctuation|capitals|twocapitals|phrases|doublespace|context on|off
#     @ reset                    back to the defaults, both lists empty
# The answers depend on the English dictionaries Windows has installed.

# ---- misspellings ----
"teh cat " => "the cat "
"i think " => "I think "
"recieve it " => "receive it "
"dont go " => "don't go "
"alot of " => "a lot of "
"improrvie " => "improve "
"hello world " => "hello world "
"npm install " => "npm install "
"kya hai teh " => "kya hai teh "

# ---- punctuation and sentence capitals ----
"teh, " => "the, "
"go now. then stop " => "go now. Then stop "
"example.com is " => "example.com is "
"see dr. smith " => "see dr. smith "
"(teh) " => "(the) "
@ punctuation off
"teh," => "teh,"
"teh, " => "the, "
@ capitals off
"go now. then stop " => "go now. then stop "
@ reset

# ---- Backspace undoes a fix ----
"teh {BS}" => "teh"
"teh {BS} " => "teh "
"teh {BS}{BS}e " => "tee "
# Backspace and then punctuation only took the space away: the fix goes back.
"reprot {BS}, then " => "report, then "
"reprot {BS}. then " => "report. Then "
# Undone twice, it stays as typed.
"reprot {BS},{BS}, then " => "reprot, then "
# Backspace and more letters: the word is being edited, and is checked again.
"reprot {BS}s then " => "reports then "

# ---- two capitals at the start of a word ----
"THe cat " => "The cat "
"HEllo " => "Hello "
"a GOod day " => "a Good day "
"THier " => "Their "
"IDs and PCs " => "IDs and PCs "
"OSes " => "OSes "
"NASA " => "NASA "
"iPhone " => "iPhone "
"McDonald " => "McDonald "
"THe {BS}" => "THe"
@ twocapitals off
"THe cat " => "THe cat "
@ reset

# ---- real words in the wrong place (data\phrase-fixes.txt) ----

"got it away form " => "got it away from "
"i wold " => "I would "

"untied states " => "United States "
"thier is " => "there is "
"and thna " => "and then "
"their is, " => "there is, "
"their is {BS}" => "their is"
"their is {BS} " => "their is "
"their is {NEXT}" => "their is "
"their is {NEXT}{NEXT}" => "there is "
# Not when something stands between the two words, or the phrase can be right.
"their, is " => "their, is "
"their. is " => "their. Is "
"could of " => "could of "
"better then " => "better then "
"fill in the form " => "fill in the form "
# "fro" becomes "for", but not in "to and fro".
"looking fro it " => "looking for it "
"to and fro " => "to and fro "
@ phrases off
"their is " => "their is "
"to and fro " => "to and for "
@ reset

# ---- smart typography (off by default) ----
"word-- next " => "word-- next "
"look... " => "look... "
"go -> there " => "go -> there "

@ smartdashes on
"word-- next " => "word— next "
"word--- next " => "word— next "
"word-- {BS}" => "word--"
"word-- could of been " => "word— could have been "
@ reset

@ smartellipsis on

"wait... {BS}" => "wait..."
"wait... could of been " => "wait… Could have been "
@ reset

@ smartarrows on

"a => b " => "a ⇒ b "
"a <- b " => "a ← b "
"a -> {BS}" => "a ->"
@ reset
